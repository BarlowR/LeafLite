# leaflite tone config — copy to the ROOT of the SD card as /tones.txt
# XCTracer XCTRACER.TXT format. Loaded once at boot; delete the file to get
# the built-in defaults back (this file ships identical to them).
#
# tone=<climb m/s>,<frequency Hz>,<cycle ms>,<duty %>
#   Up to 12 lines, climb values ascending. Output is interpolated between
#   lines and clamped past the ends. duty is the audible slice of each cycle;
#   duty=100 is a continuous tone.
# The vario is silent between SinkToneOnThreshold and ClimbToneOnThreshold.
# On/Off pairs give hysteresis (Off <= On for climb, Off >= On for sink).
# Unknown keys (setVolume, dampingFactor, ...) are ignored.

ClimbToneOnThreshold=0.10
ClimbToneOffThreshold=0.10
SinkToneOnThreshold=-4.00
SinkToneOffThreshold=-4.00
